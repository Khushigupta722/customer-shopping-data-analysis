        "\n",
        "    lower_bound = q1 - 1.5 * iqr\n",
        "    upper_bound = q3 + 1.5 * iqr\n",
        "\n",
        "    count = ((df[col] < lower_bound) | (df[col] > upper_bound)).sum()\n",
        "\n",
        "    outlier_results.append({\n",
        "        \"Column\": col,\n",
        "        \"Q1\": q1,\n",
        "        \"Q3\": q3,\n",
        "        \"IQR\": iqr,\n",
        "        \"Lower_Bound\": lower_bound,\n",
        "        \"Upper_Bound\": upper_bound,\n",
        "        \"Outlier_Count\": int(count)\n",
        "    })\n",
        "\n",
